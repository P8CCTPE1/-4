//Выполнить задание: В кафе пароль обновляется каждый час на основе двух чисел ( A и B). Новый пароль активен, если ровно одно из двух чисел четное. Запишите условия активности пароля.
//Ход работы:
//1: Подключить нужные библиотеки для корректной работы
//2: Объявить переменные A и B. 
//3: Проверяем условия: 
//Если A нечетное и B нечетное- пароль неверный
//Если A четное и B четное- пароль неверный
//Если A четное а B нечетное- пароль верный
//Если A нечетное а B четное- пароль верный
//4: Завершаем работу






#include <stdio.h>  // Библиотека для ввода/вывода (printf, scanf_s)
#include <locale.h> // Библиотека для настройки русского языка в консоли

int main() {
    // Настраиваем локаль, чтобы русский текст корректно отображался в консоли
    setlocale(LC_ALL, "Russian");

    // Объявляем переменные A и B. Используем int, так как работаем с целыми числами.
    int A, B;

    // Выводим приглашение к вводу
    printf("Введите число A: ");
    scanf_s("%d", &A); // Считываем число A

    printf("Введите число B: ");
    scanf_s("%d", &B); // Считываем число B

    // Проверяем условие: "ровно одно число четное".
    // 1. (A % 2 == 0) - проверяет, что A четное (остаток от деления на 2 равен 0)
    // 2. (B % 2 == 0) - проверяет, что B четное
    // 3. && (И) - требует выполнения обоих условий
    // 4. || (ИЛИ) - требует выполнения хотя бы одной из двух частей
    // 5. ! (НЕ) - логическое отрицание (если B % 2 != 0, это то же самое, что !(B % 2 == 0))
    
    // Условие: (A четное И B нечетное) ИЛИ (A нечетное И B четное)
    if ( ((A % 2 == 0) && (B % 2 != 0)) || ((A % 2 != 0) && (B % 2 == 0)) ) {
        printf("Пароль активен!\n"); // Если условие истинно
    } 
    else {
        printf("Пароль НЕ активен.\n"); // Если условие ложно (оба четные или оба нечетные)
    }

    return 0; // Завершаем программу с кодом успеха
}# -4
